PYTHON

def task1(arr, B):
    min_val = min(arr)
    max_val = max(arr)
    min_idx = arr.index(min_val)
    max_idx = arr.index(max_val)
    
    start = min(min_idx, max_idx)
    end = max(min_idx, max_idx)
    count = sum(1 for x in arr[start+1:end] if x > B)
    ratio = max_val / min_val if min_val != 0 else "Undefined (division by zero)"
    
    return count, ratio

# Пример использования:
A = [3, 1, 4, 1, 5, 9, 2, 6]
B = 2
print(task1(A, B))  # Вывод: (2, 9.0)

C++

#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

pair<int, float> task1(vector<int> arr, int B) {
    auto min_it = min_element(arr.begin(), arr.end());
    auto max_it = max_element(arr.begin(), arr.end());
    int min_idx = distance(arr.begin(), min_it);
    int max_idx = distance(arr.begin(), max_it);
    
    int start = min(min_idx, max_idx);
    int end = max(min_idx, max_idx);
    int count = 0;
    for (int i = start + 1; i < end; ++i) {
        if (arr[i] > B) count++;
    }
    
    float ratio = (float)*max_it / *min_it;
    return {count, ratio};
}

int main() {
    vector<int> A = {3, 1, 4, 1, 5, 9, 2, 6};
    int B = 2;
    auto result = task1(A, B);
    cout << result.first << ", " << result.second << endl;  // Вывод: 2, 9
    return 0;
}

JAVA

import java.util.Arrays;

public class Main {
    public static void main(String[] args) {
        int[] A = {3, 1, 4, 1, 5, 9, 2, 6};
        int B = 2;
        int[] result = task1(A, B);
        System.out.println(result[0] + ", " + result[1]);  // Вывод: 2, 9.0
    }

    public static int[] task1(int[] arr, int B) {
        int min = Arrays.stream(arr).min().getAsInt();
        int max = Arrays.stream(arr).max().getAsInt();
        int minIdx = 0, maxIdx = 0;
        for (int i = 0; i < arr.length; i++) {
            if (arr[i] == min) minIdx = i;
            if (arr[i] == max) maxIdx = i;
        }
        
        int start = Math.min(minIdx, maxIdx);
        int end = Math.max(minIdx, maxIdx);
        int count = 0;
        for (int i = start + 1; i < end; i++) {
            if (arr[i] > B) count++;
        }
        
        float ratio = (float)max / min;
        return new int[]{count, (int)ratio};
    }
}
